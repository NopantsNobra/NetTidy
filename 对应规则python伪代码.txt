
class FileNode:
    def __init__(self, name):
        self.name = name
        self.files = []           # 存放文件名
        self.children = []        # 存放子目录 Node
        self.ancestor_chain = []  # 记录被穿透/提级撕掉的包装纸名字，用于溯源防撞车

def clean_directory(node, is_root=False):
    """
    核心递归清理函数
    """
    # 1. 递归清理所有子目录（自底向上，保证处理当前目录时，子目录已是干净状态）
    for child in node.children:
        clean_directory(child, is_root=False)
        
    # 2. 穿透（剥洋葱）：如果子目录是空壳（0文件1子目录），用孙子目录替换它
    # 使用 while 循环是为了处理连续穿透的情况
    changed = True
    while changed:
        changed = False
        i = 0
        while i < len(node.children):
            child = node.children[i]
            if not child.files and len(child.children) == 1:
                grandchild = child.children[0]
                # 记录被撕掉的壳，用于溯源
                grandchild.ancestor_chain.append(child.name) 
                # 替换掉子目录
                node.children[i] = grandchild
                changed = True
                continue # 继续检查这个位置，因为grandchild可能也是个空壳
            i += 1

    # 3. 同生共死（分叉点判断）
    frozen = False # 冻结标记：如果触发了全留，则当前目录下禁止单文件提级
    if not node.files and len(node.children) > 1:
        # 检查是否所有子目录都只包含1个文件且无子目录
        all_single_file = all(len(c.files) == 1 and not c.children for c in node.children)
        
        if all_single_file:
            # 情况A：全提
            for child in node.children:
                for f in child.files:
                    f.ancestor_chain.append(child.name) # 记录来源
                node.files.extend(child.files)
            node.children = [] # 扔掉所有子目录
        else:
            # 情况B：全留。标记冻结，不再允许提级操作
            frozen = True
            
    # 4. 单文件提级（只有在未冻结的情况下才执行）
    if not frozen:
        i = 0
        while i < len(node.children):
            child = node.children[i]
            # 如果子目录只有1个文件，且没有更深的子目录
            if len(child.files) == 1 and not child.children:
                file = child.files[0]
                # 记录被扔掉的壳
                file.ancestor_chain.append(child.name)
                # 提至当前目录
                node.files.append(file)
                # 删除该子目录
                node.children.pop(i)
                continue
            i += 1

    # 5. 根目录锚定保护
    # 如果当前是根目录，哪怕它现在变成了0文件1子目录（空壳），也不做任何上提操作
    # 因为不可能会执行到外部去，根目录本身被天然保留
    
    return node
