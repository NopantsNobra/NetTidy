"""纯工具函数（仅依赖 constants）"""
import hashlib
import os
from .constants import (
    ILLEGAL_CHARS, WINDOWS_RESERVED, MAX_PATH_EXTENDED, IS_WINDOWS,
    ReparseType, PermissionAction,
)
# ── 路径 ──
def _win_path(path: str) -> str:
    r"""转换为 \\?\ 长路径"""
    if not IS_WINDOWS:
        return path
    path = os.path.abspath(path)
    if path.startswith("\\\\"):
        return "\\\\?\\UNC\\" + path[2:]
    return "\\\\?\\" + path
def _short_path(path: str) -> str:
    r"""从 \\?\ 还原"""
    if path.startswith("\\\\?\\UNC\\"):
        return "\\" + path[7:]
    if path.startswith("\\\\?\\"):
        return path[4:]
    return path
# ── 文件名 ──
def _sanitize_filename(name: str) -> str:
    clean = "".join(c if c not in ILLEGAL_CHARS else "_" for c in name)
    clean = clean.strip(" .")
    if not clean:
        clean = "_unnamed"
    if os.path.splitext(clean)[0].upper() in WINDOWS_RESERVED:
        clean = f"_{clean}"
    return clean
def _truncate_name(name: str, max_bytes: int) -> str:
    encoded = name.encode("utf-8")
    if len(encoded) <= max_bytes:
        return name
    base, ext = os.path.splitext(name)
    h = hashlib.sha256(name.encode("utf-8")).hexdigest()[:8]
    suffix = f"_{h}"
    avail = max_bytes - len(ext.encode("utf-8")) - len(suffix.encode("utf-8"))
    if avail < 1:
        avail = 1
    truncated = ""
    cur = 0
    for ch in base:
        cb = len(ch.encode("utf-8"))
        if cur + cb > avail:
            break
        truncated += ch
        cur += cb
    return f"{truncated or 'x'}{suffix}{ext}"
# ── 格式化 ──
def _fmt_size(size_bytes: int) -> str:
    size = size_bytes
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if size < 1024:
            return f"{size:.1f} {unit}"
        size /= 1024
    return f"{size:.1f} PB"
def _node_attrs_str(n) -> str:
    parts = []
    if getattr(n, 'is_hidden', False): parts.append("H")
    if getattr(n, 'is_readonly', False): parts.append("R")
    if getattr(n, 'is_system', False): parts.append("S")
    if getattr(n, 'has_zone_identifier', False): parts.append("Z")
    rp = getattr(n, 'reparse_type', None)
    if rp and rp != ReparseType.NONE:
        parts.append({"Symlink": "S", "Junction": "J", "HardLink": "H", "MountPoint": "M"}.get(rp.value, "?"))
    return "".join(parts) if parts else "-"
def _parse_actions(action_str: str):
    m = {
        "perm": PermissionAction.GRANT_FULL_CONTROL,

        "readonly": PermissionAction.REMOVE_READONLY, "ro": PermissionAction.REMOVE_READONLY,

        "zone": PermissionAction.CLEAR_ZONE_IDENTIFIER, "z": PermissionAction.CLEAR_ZONE_IDENTIFIER,

    }
    return {m[p.strip()] for p in action_str.lower().split(",") if p.strip() in m}

    cfg = dict(DEFAULT_CONFIG)
    if config_path and os.path.isfile(config_path):
        try:
            with open(config_path, "r", encoding="utf-8") as f:
                user = json.load(f)
            for ep in user.get("exclude_paths", []):
                if ".." in os.path.normpath(ep).split(os.sep):
                    print(f"⚠️ 配置警告: 排除路径 '{ep}' 含路径穿越，已忽略")

            cfg.update(user)
        except (json.JSONDecodeError, OSError) as e:
            print(f"⚠️ 配置文件加载失败: {e}，使用默认配置")

    errors = []
    if config.get("dedup_strategy") not in {"shallowest", "newest", "largest", "hardlink"}:
        errors.append(f"无效去重策略: {config.get('dedup_strategy')}")
    if config.get("flatten_mode") not in {"single", "ext", "date", "prefix"}:
        errors.append(f"无效扁平化模式: {config.get('flatten_mode')}")

# ── SSD/HDD 检测 ──
def detect_disk_type(drive_letter: str) -> str:
    global _disk_type_cache
    if not IS_WINDOWS:
        return "UNKNOWN"
    dl = drive_letter.upper().rstrip(":\\")
    if dl in _disk_type_cache:
        return _disk_type_cache[dl]
    try:
        cmd = [
            "powershell", "-NoProfile", "-Command",
            f"Get-PhysicalDisk | ForEach-Object {{$d=$_; Get-Partition -DiskNumber $d.DiskNumber "
            f"-ErrorAction SilentlyContinue | Where-Object {{$_.DriveLetter -eq '{dl}'}} | "
            f"Select-Object -First 1}} | ForEach-Object {{(Get-PhysicalDisk -DeviceId "
            f"$_.DiskNumber -ErrorAction SilentlyContinue).MediaType}}"
        ]
        r = subprocess.run(cmd, capture_output=True, text=True, timeout=15)
        media = r.stdout.strip()
        if "SSD" in media:
            _disk_type_cache[dl] = "SSD"
        elif "HDD" in media:
            _disk_type_cache[dl] = "HDD"
        else:
            _disk_type_cache[dl] = "UNKNOWN"
    except Exception:
        _disk_type_cache[dl] = "UNKNOWN"
    return _disk_type_cache[dl]
def apply_disk_tuning(target_path: str, config: dict) -> None:
    """根据磁盘类型自动调整 runtime 参数"""
    if not IS_WINDOWS:
        return
    drive, _ = os.path.splitdrive(os.path.abspath(target_path))
    if not drive:
        return
    dt = detect_disk_type(drive)
    manual = config.get("threads")
    if dt == "SSD":
        runtime.max_workers = manual or min(16, (os.cpu_count() or 4) * 2)
        runtime.throttle_sleep = (config.get("throttle_ms", 20)) / 1000.0
        print(f"💾 检测到 SSD ({drive}), 线程: {runtime.max_workers}, 节流: {runtime.throttle_sleep*1000:.0f}ms")
    elif dt == "HDD":
        runtime.max_workers = manual or 2
        runtime.throttle_sleep = (config.get("throttle_ms", 100)) / 1000.0
        print(f"💿 检测到 HDD ({drive}), 线程: {runtime.max_workers}, 节流: {runtime.throttle_sleep*1000:.0f}ms")
    else:
        runtime.max_workers = manual or DEFAULT_MAX_WORKERS
        runtime.throttle_sleep = (config.get("throttle_ms", 50)) / 1000.0
        print(f"💿 磁盘未知 ({drive}), 线程: {runtime.max_workers}, 节流: {runtime.throttle_sleep*1000:.0f}ms")
def update_workers(n: int) -> None:
    runtime.max_workers = max(1, n)

    def quick_hash(self) -> str:
        if self._quick_hash is not None:
            return self._quick_hash
        if self.is_dir:
            self._quick_hash = ""
            return ""
        try:
            wp = _win_path(self.full_path)
            fp = wp if os.path.exists(wp) else self.full_path
            with open(fp, "rb") as f:
                self._quick_hash = hashlib.sha256(f.read(4096)).hexdigest()
        except (OSError, IOError):
            self._quick_hash = ""
        return self._quick_hash
    def full_hash(self) -> str:
        if self._full_hash is not None:
            return self._full_hash
        if self.is_dir:
            self._full_hash = ""
            return ""
        try:
            wp = _win_path(self.full_path)
            fp = wp if os.path.exists(wp) else self.full_path
            h = hashlib.sha256()
            with open(fp, "rb") as f:
                while chunk := f.read(65536):
                    h.update(chunk)
            self._full_hash = h.hexdigest()
        except (OSError, IOError):
            self._full_hash = ""
        return self._full_hash

"""Win32 API 底层 IO 封装 + Windows 系统回收站集成"""
import ctypes
import os
import shutil
from typing import Optional, List, Tuple
from .constants import (
    IS_WINDOWS, MOVEFILE_COPY_ALLOWED, MOVEFILE_REPLACE_EXISTING,
    MOVEFILE_WRITE_THROUGH, FILE_ATTRIBUTE_HIDDEN, FILE_ATTRIBUTE_READONLY,
    FILE_ATTRIBUTE_SYSTEM, FILE_ATTRIBUTE_REPARSE, FILE_ATTRIBUTE_DIRECTORY,
    IO_REPARSE_TAG_SYMLINK, IO_REPARSE_TAG_MOUNT_POINT,
    FO_DELETE, FOF_ALLOWUNDO, FOF_NOCONFIRMATION, FOF_SILENT, FOF_NOERRORUI,
)
from .utils import _win_path, _short_path
if IS_WINDOWS:
    from ctypes import wintypes
    _kernel32 = ctypes.windll.kernel32
    _shell32 = ctypes.windll.shell32
class Win32IO:
    """Win32 API 底层 IO，完整支持长路径"""
    @classmethod
    def get_file_attributes(cls, path: str) -> Optional[int]:
        if not IS_WINDOWS:
            return None
        for p in (_win_path(path), path):
            attrs = _kernel32.GetFileAttributesW(p)
            if attrs != -1:
                return attrs
        return None
    @classmethod
    def exists(cls, path: str) -> bool:
        if not IS_WINDOWS:
            return os.path.exists(path)
        return cls.get_file_attributes(path) is not None
    @classmethod
    def is_dir(cls, path: str) -> bool:
        if not IS_WINDOWS:
            return os.path.isdir(path)
        attrs = cls.get_file_attributes(path)
        return bool(attrs & FILE_ATTRIBUTE_DIRECTORY) if attrs is not None else False
    @classmethod
    def safe_stat(cls, path):
        try:
            wp = _win_path(path)
            if os.path.exists(wp):
                return os.lstat(wp)
        except (OSError, ValueError):
            pass
        try:
            return os.lstat(path)
        except (OSError, ValueError):
            return None
    @classmethod
    def safe_listdir(cls, path: str) -> List[str]:
        for p in (_win_path(path), path):
            try:
                if cls.is_dir(path):
                    return os.listdir(p)
            except (OSError, ValueError):
                continue
        return []
    @classmethod
    def move(cls, src: str, dst: str) -> bool:
        if not IS_WINDOWS:
            try:
                shutil.move(src, dst)
                return True
            except OSError:
                return False
        for s, d in [(_win_path(src), _win_path(dst)), (src, dst)]:
            if _kernel32.MoveFileExW(s, d,
                    MOVEFILE_COPY_ALLOWED | MOVEFILE_REPLACE_EXISTING | MOVEFILE_WRITE_THROUGH):
                return True
        return False
    @classmethod
    def delete_file(cls, path: str) -> bool:
        if not IS_WINDOWS:
            try:
                os.remove(path)
                return True
            except OSError:
                return False
        for p in (_win_path(path), path):
            if _kernel32.DeleteFileW(p):
                return True
        return False
    @classmethod
    def remove_directory(cls, path: str) -> bool:
        if not IS_WINDOWS:
            try:
                os.rmdir(path)
                return True
            except OSError:
                return False
        for p in (_win_path(path), path):
            if _kernel32.RemoveDirectoryW(p):
                return True
        return False
    @classmethod
    def get_reparse_tag(cls, path: str) -> Optional[int]:
        if not IS_WINDOWS:
            return None
        try:
            import struct
            wp = _win_path(path)
            handle = _kernel32.CreateFileW(
                wp, 0x80000000, 0x00000001 | 0x00000002 | 0x00000004,
                None, 3, 0x00200000 | 0x02000000, None)
            if handle == -1 or handle is None:
                return None
            try:
                buf = ctypes.create_string_buffer(1024)
                br = wintypes.DWORD()
                if _kernel32.DeviceIoControl(handle, 0x000900A8, None, 0, buf, 1024,

                    return struct.unpack_from("<I", buf.raw, 0)[0]
            finally:
                _kernel32.CloseHandle(handle)
        except Exception:
            pass

class RecycleBin:
    """
    Windows 系统回收站操作
    使用 SHFileOperationW API 将文件/目录移入系统回收站。
    兼容 Windows 7+，零外部依赖。
    与自定义 .restructure_trash/ 的区别：
    ┌──────────────┬─────────────────────┬─────────────────────┐
    │              │ .restructure_trash  │ 系统回收站            │
    ├──────────────┼─────────────────────┼─────────────────────┤
    │ 脚本自动回滚  │ ✅ 支持              │ ❌ 不支持            │
    │ 用户习惯     │ 需要手动找目录       │ ✅ 回收站图标恢复     │
    │ 磁盘空间     │ 占用目标卷空间       │ 不额外占用           │
    │ 磁盘配额     │ 无感知              │ ⚠️ 受回收站配额限制   │
    │ 超长路径     │ ✅ \\?\ 支持        │ ⚠️ 部分受限          │
    └──────────────┴─────────────────────┴─────────────────────┘
    建议：默认使用 .restructure_trash/，仅在确认不需要脚本回滚时使用 --use-recycle-bin
    """
    @classmethod
    def move_to_bin(cls, path: str, silent: bool = True) -> Tuple[bool, str]:
        """
        将文件或目录移入 Windows 系统回收站
        Args:
            path: 要删除的文件或目录路径
            silent: 是否静默（不弹出确认对话框）
        Returns:
            (success: bool, error_message: str)
            成功时 error_message 为空字符串
        """
        if not IS_WINDOWS:
            return False, "仅支持 Windows 系统"
        # 路径安全检查
        path = os.path.abspath(path)
        if not os.path.exists(path) and not Win32IO.exists(path):
            return False, "文件或目录不存在"
        # SHFileOperationW 需要 pFrom 以双 null 终止
        # 对于单个路径: "C:\\path\\file.txt\0\0"
        # 长路径前缀 \\?\ 可被 SHFileOperationW 识别 (Windows Vista+)
        path_to_use = _win_path(path)
        path_buffer = path_to_use + '\0'
        op = _SHFILEOPSTRUCTW()
        op.wFunc = FO_DELETE
        op.pFrom = path_buffer
        op.pTo = None
        op.hNameMappings = None
        op.lpszProgressTitle = None
        flags = FOF_ALLOWUNDO  # 关键标志：放入回收站而非永久删除
        if silent:
            flags |= FOF_NOCONFIRMATION | FOF_SILENT | FOF_NOERRORUI
        op.fFlags = flags
        try:
            result = _shell32.SHFileOperationW(ctypes.byref(op))
        except OSError as e:
            return False, f"SHFileOperation 调用异常: {e}"
        if result == 0:
            if op.fAnyOperationsAborted:
                return False, "操作被用户取消"
            return True, ""
        else:
            msg = _RECYCLE_ERROR_MAP.get(result, f"SHFileOperation 错误码: 0x{result:08X}")
            # 额外提示：回收站满时的建议
            if result == 0x0007:
                msg += " (提示: 可尝试清空回收站或增加回收站容量)"
            return False, msg
    @classmethod
    def check_support(cls, path: str = None) -> Tuple[bool, str]:
        """
        检测当前环境是否支持系统回收站操作
        Returns:
            (supported, reason)
        """
        if not IS_WINDOWS:
            return False, "非 Windows 系统"
        try:
            if not hasattr(cls, '_shell32_checked'):
                cls._shell32_checked = True
                if not hasattr(ctypes.windll, 'shell32'):
                    return False, "shell32.dll 不可用"
            # 尝试在临时目录做一次空操作来验证 API 可用性
            return True, ""
        except Exception as e:
            return False, str(e)

class SafetyChecker:
    @classmethod

        dirs = set()
        for v in ("SystemRoot", "ProgramFiles", "ProgramFiles(x86)", "ProgramData"):
            val = os.environ.get(v, "")
            if val:
                dirs.add(val)
        for dl in "CDEFGHIJKLMNOPQRSTUVWXYZ":
            root = f"{dl}:\\"
            if os.path.exists(root):
                for sub in ("$Recycle.Bin", "System Volume Information",
                            "Windows", "Program Files", "Program Files (x86)", "ProgramData"):
                    dirs.add(os.path.join(root, sub))
        return dirs
    @classmethod
    def _user_dirs(cls) -> Set[str]:
        dirs = set()
        up = os.environ.get("USERPROFILE", "")
        if up:
            for sub in ("AppData", os.path.join("AppData", "Local"),
                        os.path.join("AppData", "Roaming"), os.path.join("AppData", "LocalLow")):
                dirs.add(os.path.join(up, sub))
        return dirs
    @classmethod
    def check(cls, target_path: str) -> Tuple[SafetyLevel, str]:
        tp = os.path.abspath(target_path)
        if not os.path.exists(tp):
            return SafetyLevel.BLOCKED, f"路径不存在: {tp}"
        drive, _ = os.path.splitdrive(tp)
        if tp in (drive + "\\", drive + "/"):
            return SafetyLevel.BLOCKED, "禁止操作驱动器根目录"
        protected = cls._system_dirs() | cls._user_dirs()
        tl = tp.lower()
        for sd in protected:
            sl = sd.lower()
            if tl == sl or tl.startswith(sl + "\\"):
                return SafetyLevel.BLOCKED, f"命中系统保护目录: {sd}"
        if not os.access(tp, os.W_OK):
            return SafetyLevel.RESTRICTED, "目标目录无写入权限"
        try:
            if shutil.disk_usage(tp).free < 200 * 1024 * 1024:
                return SafetyLevel.RESTRICTED, "磁盘剩余空间不足 200MB"
        except OSError:
            pass
        test = os.path.join(tp, f".restr_test_{os.getpid()}")
        try:
            with open(test, "w") as f:
                f.write("x")
            os.remove(test)
        except OSError:
            return SafetyLevel.RESTRICTED, "目标文件系统为只读"
        return SafetyLevel.FULL, "安全"

        items = targets or self.nodes
        user = os.environ.get("USERNAME", "")
        if not user:
            print("⚠️ 无法获取用户名，跳过"); return self.stats

                                  capture_output=True, text=True, timeout=30)

                    node.status = NodeStatus.PERM_FIXED; self.stats["perm_granted"] += 1
                else:

            except subprocess.TimeoutExpired:

            except FileNotFoundError:
                print("❌ icacls 不可用"); return self.stats

        items = targets or [n for n in self.nodes if n.is_readonly]

                node.is_readonly = False; node.status = NodeStatus.ATTR_FIXED; self.stats["readonly_removed"] += 1
            except (OSError, subprocess.TimeoutExpired) as e:

        items = targets or []
        if not targets:
            for n in self.nodes:
                if n.is_dir or n.action.value == "DELETE": continue
                if ZoneIdentifierHandler.has_zone_identifier(n.full_path):
                    n.has_zone_identifier = True; items.append(n)

            else:
                self._errors.add(node.full_path, "ZoneFail", ""); self.stats["failed"] += 1

        if PermissionAction.GRANT_FULL_CONTROL in actions or PermissionAction.ALL in actions:

        if PermissionAction.CLEAR_ZONE_IDENTIFIER in actions or PermissionAction.ALL in actions:
            self.clear_zone_identifiers(dry_run=dry_run)

class FileListParser:

    def __init__(self):
        self.total_truncated = 0
        self.total_skipped = 0
        self._tm = {}
        self._dm = {}
        self._hcf = False
    def load_files(self, fps):
        all_l = []; self.total_truncated = self.total_skipped = 0
        self._tm = {}; self._dm = {}; self._hcf = False
        for fp in fps:
            if not os.path.isfile(fp): print(f"❌ 不存在: {fp}"); continue
            l, tc, sc = self._load1(fp); all_l.extend(l)
            self.total_truncated += tc; self.total_skipped += sc
        b = len(all_l); all_l = list(dict.fromkeys(all_l))
        if len(all_l) < b: print(f"  去重: {b}→{len(all_l)}")
        return all_l
    def _load1(self, fp):
        with open(fp, "r", encoding="utf-8-sig") as f: raw = f.readlines()
        valid = []; tc = sc = 0; inh = True; fn = False
        for r in raw:
            ln = r.rstrip("\r\n")
            if not ln.strip(): sc += 1; continue
            m = self._CRE.match(ln)
            if m:
                self._hcf = True; self._tm[m.group(2).strip()] = m.group(3) == "DIR"
                self._dm[m.group(2).strip()] = int(m.group(1)); valid.append(m.group(2).strip()); continue
            if inh:
                s = ln.strip()
                if s == "FullName" or "FullName" in s: fn = True; sc += 1; continue
                if s.startswith("-" * 8): inh = False; sc += 1; continue
                inh = False
            if TRUNCATED_LINE_RE.search(ln.rstrip()): tc += 1; continue
            valid.append(ln)
        return valid, tc, sc
    def parse(self, lines):
        S = "\\"
        uc = self._hcf and bool(self._tm)
        rp = None
        for ln in lines:
            ps = ln.split(S)
            if rp is None: rp = list(ps)
            else:
                nr = []
                for a, b in zip(rp, ps):
                    if a == b: nr.append(a)
                    else: break
                rp = nr
        root = S.join(rp) if rp else ""
        if re.match(r'^[A-Z]:$', root, re.I): root += S
        rd = len(rp) if rp else 1
        aint = set()
        for ln in lines:
            ps = ln.split(S)
            for i in range(rd, len(ps) - 1): aint.add(S.join(ps[:i + 1]))
        sp = sorted(aint | set(lines), key=lambda p: p.lower())
        def hc(p):
            px = p.lower() + S; lo = bisect.bisect_left(sp, px, key=lambda x: x.lower())
            return lo < len(sp) and sp[lo].lower().startswith(px)
        idm = {}
        for ln in lines:
            if ln in aint: idm[ln] = True
            elif uc and ln in self._tm: idm[ln] = self._tm[ln]
            elif hc(ln): idm[ln] = True
            else: idm[ln] = False
        nodes = []; ex = set(); imp = 0
        for ln in lines:
            ps = ln.split(S)
            rld = self._dm.get(ln, len(ps) - rd) if uc else len(ps) - rd
            fn = ps[-1] if ps else ""; id = idm.get(ln, False)
            ext = "" if id else ("." + fn.rsplit(".", 1)[1].lower() if "." in fn else "")
            nodes.append(VirtualFileNode(ln, fn, ext, depth=rld, is_dir=id)); ex.add(ln)
        for dp in aint:
            if dp not in ex:
                dps = dp.split(S)
                nodes.append(VirtualFileNode(dp, dps[-1], "", depth=len(dps) - rd, is_dir=True)); imp += 1
        st = {"root": root, "total_input_lines": len(lines),
              "dirs_detected": sum(1 for v in idm.values() if v),
              "files_detected": sum(1 for v in idm.values() if not v),
              "implicit_dirs": imp, "custom_format": uc}
        return nodes, root, st

            for j in range(lo, len(sn)):
                if sn[j].full_path.lower().startswith(pfx):
                    if sn[j].action == NodeAction.NONE: sn[j].action = NodeAction.DELETE; sn[j].reason = f"子树:{root.file_name}"
                else: break
    def deduplicate_by_name(self):
        files = [n for n in self.nodes if not n.is_dir and n.action == NodeAction.NONE]
        ng: Dict[str, list] = defaultdict(list)
        for f in files: ng[f.file_name.lower()].append(f)
        dc = 0
        for name, g in ng.items():
            if len(g) < 2: continue
            hs = any(n.size_bytes > 0 for n in g)
            if hs:
                ssg: Dict[int, list] = defaultdict(list)
                for n in g: ssg[n.size_bytes].append(n)
                for sz, sg2 in ssg.items():
                    if len(sg2) < 2: continue
                    keep = min(sg2, key=lambda n: n.depth)
                    self.duplicate_groups.append({"name": name, "size": sz, "keep": keep.full_path,
                                                  "delete": [n.full_path for n in sg2 if n.full_path != keep.full_path]})
                    for n in sg2:
                        if n.full_path != keep.full_path:
                            n.action = NodeAction.DELETE; n.reason = f"同名重复(d{keep.depth})"; dc += 1
            else:
                keep = min(g, key=lambda n: n.depth)
                for n in g: n.reason = "同名疑似重复"
                self.duplicate_groups.append({"name": name, "size": None, "keep": keep.full_path,
                                              "delete": [m.full_path for m in g if m.full_path != keep.full_path]})
        return dc
class OfflineRestructureEngine:
    def __init__(self, nodes, root):
        self.nodes = nodes; self.root = root
    def unpack_redundant(self):
        S = "\\"; up = 0
        dirs = [n for n in self.nodes if n.is_dir and n.action == NodeAction.NONE]
        dc = defaultdict(list)
        for d in dirs: dc[S.join(d.full_path.split(S)[:-1]).lower()].append(d)
        for pl, ch in dc.items():
            if len(ch) != 1: continue
            child = ch[0]; pn = pl.split(S)[-1]
            if pl != self.root.lower() else "":
                pass
            if not pn or pn != child.file_name.lower(): continue
            ap = S.join(child.full_path.split(S)[:-1]); cp = child.full_path.lower() + S
            for n in self.nodes:
                if n.full_path.lower().startswith(cp) and n.action == NodeAction.NONE:
                    rel = n.full_path[len(child.full_path):].lstrip(S)
                    n.target_path = S.join([ap, rel]) if rel else ap; n.reason = f"解包:{child.file_name}"; n.action = NodeAction.MOVE
            child.action = NodeAction.DELETE; child.reason = "冗余目录"; up += 1
        return up
def print_offline_report(stats, ps, cc, jc, dc, dg, up, nodes, tc, sc, fc=0):

        if args.flatten is not None or args.only_flatten:
            ft = args.flatten if args.flatten else os.path.join(engine.target_path, "_flattened")
            print(f"\n📦 扁平化: {ft} (模式:{fm})")
            FlattenEngine(engine.nodes, engine.target_path, engine.errors,
                         config.get("ext_group_map")).extract_files(ft, mode=fm)
        dup_groups = engine.get_duplicate_preview()
        if dup_groups:
            print(f"\n📋 重复组 ({len(dup_groups)}):")
            for g in dup_groups[:20]:
                al = "🔗" if g.get("action") == "HARDLINK" else "🗑️"
                print(f"  {g['group_id']}: {g['size']}\n    ✅ {g['keep']['path']}")
                for d in g["delete"]: print(f"    {al} {d['path']}")
        # 权限
        ps = {}
        if args.fix_attr or args.only_perm:
            pa = _parse_actions(args.fix_attr) if args.fix_attr else {PermissionAction.ALL}
            if pa: ps = engine.fix_permissions(pa, dry_run=not args.execute)
        # 交互式
        if args.preview_interactive:
            pv = engine.preview()
            if not pv: print("\n✅ 无需操作"); return
            sp = engine.get_space_estimate()
            print(f"\n💡 空间预估: {sp['total_mb']} MB")
            InteractivePreview(engine.nodes, dup_groups).run_apply(input)  # noqa: will call run() internally
            # 正确方式:
            ip = InteractivePreview(engine.nodes, dup_groups)
            sel = ip.run()
            ip.apply_selection(sel)
            pv = engine.preview()
            if not pv: print("✅ 已取消"); return
            sp = engine.get_space_estimate()
            print(f"\n💡 选择后: {sp['total_mb']} MB")
            if not args.execute: print("\n💡 加 --execute 执行"); return
        pv = engine.preview()
        if pv and not args.preview_interactive:
            print(f"\n📋 变更 ({len(pv)} 项):\n" + "-" * 65)
            for i, it in enumerate(pv[:50], 1):
                ic = {"DELETE": "🗑️", "MOVE": "📦", "HARDLINK": "🔗"}.get(it["action"], "?")
                if it["action"] in ("DELETE", "HARDLINK"):
                    print(f"  {i:4d}. {ic} {it['action']} {it['source']} ({it['size']})")
                else:
                    print(f"  {i:4d}. {ic} {it['source']}\n        → {it['target']}")
            if len(pv) > 50: print(f"  ... +{len(pv)-50}")
            print("-" * 65)
            sp = engine.get_space_estimate()
            if sp['total_mb'] > 0:
                print(f"\n💾 预计释放: {sp['delete_mb']}MB (删) + {sp['hardlink_mb']}MB (链) = {sp['total_mb']}MB")
        if args.preview: print("\n💡 加 --execute 执行"); return
        if args.execute:
            tot = len(pv) + sum(v for k, v in ps.items() if k not in ("failed", "skipped"))
            if not args.yes and tot > 0:
                sp = engine.get_space_estimate()
                msg = f"\n⚠️ 即将执行 {tot} 项"
                if sp['total_mb'] > 0: msg += f" (释放 {sp['total_mb']}MB)"
                msg += "。确认?
                if c.lower() not in ("yes", "y"): print("❌ 已取消"); return
            engine.execute(dry_run=False)
            if args.html_report:
                stats = engine.diagnostic.get_stats() if engine.diagnostic else {}
                sp = engine.get_space_estimate()
                HTMLReportGenerator().generate(stats, engine.nodes, dup_groups,
                                               sp.get('total_bytes', 0), args.html_report)
                print(f"\n📊 HTML: {args.html_report}")
    except KeyboardInterrupt:
        print("\n\n⚠️ 中断")
    finally:
        if locked: lock.release()

**Windows 真回收站集成说明**
以下是关于 `--use-recycle-bin` 的完整技术说明：
```
┌──────────────────────────────────────────────────────────┐
│                系统回收站 vs 自定义回收站                   │
├─────────────┬──────────────────┬────────────────────────┤
│             │ .restructure_    │ Windows 系统回收站      │
│             │ trash/           │ (--use-recycle-bin)    │
├─────────────┼──────────────────┼────────────────────────┤
│ 默认        │ ✅ 是             │ ❌ 需显式启用           │
│ 脚本回滚    │ ✅ rollback.py   │ ❌ 不支持              │
│ 用户恢复    │ 手动找目录       │ ✅ 回收站图标右键恢复   │
│ 额外空间    │ 占用目标卷       │ 不额外占用             │
│ 配额限制    │ 无               │ ⚠️ 受回收站容量限制     │
│ 长路径      │ ✅ \\?\ 完整支持  │ ⚠️ SHFileOperation    │
│             │                  │   对超长路径可能失败     │
│ 安全降级    │ —                │ ✅ 失败自动降级到       │
│             │                  │   .restructure_trash/  │
│ 回滚脚本    │ shutil.move      │ 标注"需手动恢复"       │
│             │                  │ 跳过该条目              │
└─────────────┴──────────────────┴────────────────────────┘
```
**使用方式：**
```bash
# 默认（自定义回收站，支持脚本回滚）
python -m folder_restructure_v8 D:\Projects --execute
# 启用系统回收站（用户体验更自然）
python -m folder_restructure_v8 D:\Projects --use-recycle-bin --execute
# 配置文件方式
# restructure_config.json: { "use_recycle_bin": true }
python -m folder_restructure_v8 D:\Projects --config restructure_config.json --execute
```
**安全降级机制**：当 `SHFileOperationW` 返回错误（如回收站已满、路径过长、文件被占用等），引擎自动降级到 `.restructure_trash/`，并在错误日志中记录原因。不会因为系统回收站失败而丢失文件。
**实现原理**：使用 `shell32.SHFileOperationW` API，设置 `FOF_ALLOWUNDO` 标志。该 API 从 Windows 2000 起可用，兼容性极佳。路径使用 `\\?\` 前缀以支持长路径（Windows Vista+）。错误码有完整的中文映射，方便排查。
