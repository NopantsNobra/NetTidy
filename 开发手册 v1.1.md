## 第一章 开发总览
### 1.1 开发手册定位
本文档面向开发者，提供从系统架构、模块设计、算法实现到并发控制的完整技术规范。它是产品手册《理巢 v1.1》的工程实现指南，所有设计决策均以《精确规则.md》为绝对准则。
### 1.2 核心开发原则
1. **算法与IO彻底分离**：树结构构建、穿透判定、冲突重命名等核心逻辑在内存中完成，绝不直接操作磁盘文件系统。
2. **确定性优先**：给定相同输入，必定产生完全相同的映射表和目标结构，不依赖随机数或系统状态。
3. **防御性编程**：对文件系统异常（权限不足、文件占用、路径超长）进行多层防护，参考 V11.0 的 IO 重试与错误收集机制。
4. **可观测性**：通过结构化日志、进度条、HTML 报告，让用户和开发者全程掌握系统状态。
### 1.3 与 V11.0 的技术借鉴
从《产品功能手册_V11.0.md》中，我们借鉴并改造了以下技术点：
- **五层安全防护**：系统目录保护、路径安全检查、软删除回收站（理巢采用 DRY RUN 预审流替代直接删除，更安全）、回滚脚本、锁文件机制。
- **IO 重试机制**：对文件复制/移动操作进行 3 次重试，间隔 500ms，应对杀毒软件或资源管理器临时锁定。
- **性能自适应**：借鉴 SSD/HDD 自动检测，动态调整线程数与节流间隔。
- **线程安全错误收集**：使用 `ErrorCollector` 收集多线程环境下的异常，不中断整体流程。
- **JSON 格式回滚脚本**：使用 `json.dumps()` 处理路径，确保跨 Python 版本兼容性。

### 3.1 FileNode (文件系统内存镜像)
```python
from dataclasses import dataclass, field
from pathlib import Path
from typing import List
@dataclass
class FileNode:
    """文件系统的内存镜像节点，是算法操作的唯一对象"""
    name: str                # 文件/文件夹名（保留原始大小写）
    path: Path               # 绝对路径（用于日志，算法中不直接比较）
    is_file: bool            # 是否为文件
    file_size: int = 0       # 文件大小（字节）
    children: List['FileNode'] = field(default_factory=list)
    
    # --- 算法运行时动态字段 ---
    sw: int = 0              # 穿透深度，初始为0
    tier: int = 0            # 档位: 0=未判定, 1=纯空壳, 2=分支型终点, 3=单文件叶, 4=文件型终点
    ancestry_chain: List[str] = field(default_factory=list)  # 穿透血统链，记录被穿透消失的祖先名
    
    # --- 重复检测标记 ---
    is_duplicate: bool = False
    duplicate_of: str = ""   # 重复源的路径
    
    # --- 文件名清洗后字段 ---
    safe_name: str = ""      # 经过 Windows 合规清洗后的名称
    
    def __post_init__(self):
        if not self.safe_name:
            self.safe_name = self.name
```

### 4.2 算法主函数：`resolve()`
```python
# core/penetrator.py
def resolve(node: FileNode, sw: int = 0, chain: List[str] = None) -> ResolveResult:
    """
    递归判定节点，返回处理结果。
    
    参数:
        node: 当前节点
        sw: 当前穿透深度（从上层传入）
        chain: 穿透血统链（记录被穿透消失的祖先名）
    
    返回:
        ResolveResult
    """
    if chain is None:
        chain = []
    
    # Step 1: 过滤系统文件，统计有效内容
    valid_children = system_filter.filter(node.children)
    f_count = sum(1 for c in valid_children if c.is_file)
    d_count = sum(1 for c in valid_children if not c.is_file)
    
    # Step 2: 四档分流
    if f_count == 0 and d_count == 1:
        # 第一档：纯空壳节点
        child = valid_children[0]
        child.sw = sw + 1
        child.ancestry_chain = chain + [node.name]
        return resolve(child, sw + 1, chain + [node.name])
    
    elif f_count == 0 and d_count >= 2:
        # 第二档：分支型终点
        node.tier = 2
        node.sw = sw
        node.ancestry_chain = chain
        
        if sw == 0:
            # 原地保留，处理子项
            processed_children = []
            for child in valid_children:
                result = resolve(child, sw=0, chain=[])
                if result.action == "EXTRACT_FILE":
                    # 单文件叶提取上来的文件
                    processed_children.append(result.extracted_file)
                elif result.action == "KEEP":
                    processed_children.append(result.node)
                # DISAPPEAR 的节点直接丢弃
            
            # 执行同生共死检查
            processed_children = co_extractor.check(node, processed_children)
            
            # 冲突重命名
            processed_children = renamer.rename_conflicts(processed_children)
            
            node.children = processed_children
            return ResolveResult(node=node, children=processed_children, action="KEEP")
        else:
            # sw > 0，向上抛出
            return ResolveResult(node=node, children=[], action="KEEP")
    
    elif f_count == 1 and d_count == 0:
        # 第三档：单文件叶节点
        node.tier = 3
        file_node = valid_children[0]
        file_node.ancestry_chain = chain + [node.name]
        return ResolveResult(node=None, children=[], action="EXTRACT_FILE", extracted_file=file_node)
    
    else:
        # 第四档：文件型终点 (f_count >= 2 或 f_count >= 1 and d_count >= 1)
        node.tier = 4
        node.sw = sw
        node.ancestry_chain = chain
        
        if sw == 0:
            # 原地保留，处理子项
            processed_children = []
            for child in valid_children:
                result = resolve(child, sw=0, chain=[])
                if result.action == "EXTRACT_FILE":
                    processed_children.append(result.extracted_file)
                elif result.action == "KEEP":
                    processed_children.append(result.node)
            
            # 冲突重命名
            processed_children = renamer.rename_conflicts(processed_children)
            
            node.children = processed_children
            return ResolveResult(node=node, children=processed_children, action="KEEP")
        else:
            # sw > 0，向上抛出
            return ResolveResult(node=node, children=[], action="KEEP")
```
### 4.3 同生共死检查器：`co_extractor.check()`
```python
# core/co_extractor.py
def check(parent_node: FileNode, children: List[FileNode]) -> List[FileNode]:
    """
    仅对第二档节点 (sw=0) 执行。
    
    检查子项中是否有 sw>0 且为第四档（文件型终点）的单文件子项。
    若全部满足，则提取文件，子文件夹消失。
    """
    extract_pool = [
        c for c in children 
        if c.sw > 0 and c.tier == 4 and len([ch for ch in c.children if ch.is_file]) == 1
    ]
    
    can_extract = len(extract_pool) > 0 and all(
        len([ch for ch in c.children if ch.is_file]) == 1 
        for c in extract_pool
    )
    
    if can_extract:
        extracted_files = []
        for folder in extract_pool:
            # 提取唯一文件
            file_node = next(ch for ch in folder.children if ch.is_file)
            file_node.ancestry_chain = folder.ancestry_chain + [folder.name]
            extracted_files.append(file_node)
        
        # 保留非提取池的子项
        kept_children = [c for c in children if c not in extract_pool]
        return kept_children + extracted_files
    else:
        return children
```
### 4.4 冲突重命名器：`renamer.rename_conflicts()`
```python
# core/renamer.py
def rename_conflicts(items: List[FileNode]) -> List[FileNode]:
    """
    统一溯源前缀法解决重名冲突。
    """
    from collections import defaultdict
    import re
    
    # 按名称（大小写不敏感）分组
    groups = defaultdict(list)
    for item in items:
        groups[item.safe_name.lower()].append(item)
    
    result = []
    for name_lower, conflict_items in groups.items():
        if len(conflict_items) == 1:
            result.append(conflict_items[0])
            continue
        
        # 发生冲突，对每个项溯源前缀
        renamed_items = []
        for item in conflict_items:
            prefix = find_first_different_ancestor(item, conflict_items)
            if prefix:
                # 清洗前缀
                safe_prefix = name_sanitizer.sanitize_prefix(prefix)
                # 拼接新名
                if item.is_file:
                    stem = item.safe_name.stem
                    suffix = item.safe_name.suffix
                    new_name = f"{safe_prefix}-{stem}{suffix}"
                else:
                    new_name = f"{safe_prefix}-{item.safe_name}"
                item.safe_name = name_sanitizer.sanitize_full(new_name)
            renamed_items.append(item)
        
        # 二次冲突检查（递归处理极罕见情况）
        result.extend(rename_conflicts(renamed_items))
    
    return result
def find_first_different_ancestor(item: FileNode, conflict_items: List[FileNode]) -> str:
    """
    沿 ancestry_chain 向上找第一个与其他冲突项不同的祖先名。
    """
    for i, ancestor_name in enumerate(item.ancestry_chain):
        for other in conflict_items:
            if other is item:
                continue
            # 检查其他项在位置 i 是否有相同祖先
            if i >= len(other.ancestry_chain) or not names_equal(ancestor_name, other.ancestry_chain[i]):
                return ancestor_name
    return None
```

```python
# core/scanner.py
class Scanner:
    def __init__(self, source_path: Path, config: dict):
        self.source = source_path
        self.config = config
        self.system_filter = SystemFilter()
    
    def scan(self) -> FileNode:
        """递归构建源目录树"""
        root_node = self._scan_node(self.source, depth=0)
        return root_node
    
    def _scan_node(self, path: Path, depth: int) -> FileNode:
        if depth > self.config.get('max_depth', float('inf')):
            return None
        
        try:
            entries = list(path.iterdir())
        except PermissionError:
            logger.warning(f"跳过无权限目录: {path}")
            return None
        
        children = []
        for entry in entries:
            # 跳过系统文件
            if self.system_filter.is_system_file(entry.name):
                continue
            
            if entry.is_file():
                node = FileNode(
                    name=entry.name,
                    path=entry,
                    is_file=True,
                    file_size=entry.stat().st_size
                )
                children.append(node)
            elif entry.is_dir():
                child_node = self._scan_node(entry, depth + 1)
                if child_node:  # 非空目录才添加
                    children.append(child_node)
        
        return FileNode(
            name=path.name,
            path=path,
            is_file=False,
            children=children
        )
```
### 5.2 文件名清洗器
```python
# safety/name_sanitizer.py
class NameSanitizer:
    # Windows 非法字符
    ILLEGAL_CHARS = r'[<>:"/\\|?*]'
    # Windows 保留名称
    RESERVED_NAMES = {'CON', 'PRN', 'AUX', 'NUL'} | {f'COM{i}' for i in range(1, 10)} | {f'LPT{i}' for i in range(1, 10)}
    
    def sanitize_prefix(self, prefix: str) -> str:
        """清洗溯源前缀（祖先名）"""
        prefix = re.sub(self.ILLEGAL_CHARS, '_', prefix)
        prefix = prefix.rstrip('. ')
        if prefix.upper() in self.RESERVED_NAMES:
            prefix += '_'
        return prefix
    
    def sanitize_full(self, name: str) -> str:
        """清洗完整文件名"""
        # 分离扩展名
        if '.' in name:
            stem, suffix = name.rsplit('.', 1)
            stem = self._sanitize_stem(stem)
            return f"{stem}.{suffix}"
        else:
            return self._sanitize_stem(name)
    
    def _sanitize_stem(self, stem: str) -> str:
        stem = re.sub(self.ILLEGAL_CHARS, '_', stem)
        stem = stem.rstrip('. ')
        if stem.upper() in self.RESERVED_NAMES:
            stem += '_'
        if not stem:
            stem = '_unnamed'
        return stem
    
    def truncate_path(self, path: Path, max_length: int = 259) -> Path:
        """智能截断超长路径，保留扩展名"""
        path_str = str(path)
        if len(path_str) <= max_length:
            return path
        
        # 提取扩展名
        suffix = path.suffix
        stem = path.stem
        
        # 计算可用长度
        parent_str = str(path.parent)
        available = max_length - len(parent_str) - len(suffix) - 1  # -1 for backslash
        
        if available <= 0:
            # 极端情况，只保留扩展名
            return path.parent / f"_truncated{suffix}"
        
        truncated_stem = stem[:available]
        return path.parent / f"{truncated_stem}{suffix}"
```
### 5.3 重复文件检测器 (四级哈希)
```python
# enhance/dup_detector.py
class DupDetector:
    def __init__(self, map_entries: List[MapEntry], strategy: str = 'keep_first'):
        self.entries = map_entries
        self.strategy = strategy
    
    def detect(self) -> List[MapEntry]:
        """执行四级哈希去重"""
        # 第一级：按大小分组
        size_groups = defaultdict(list)
        for entry in self.entries:
            if entry.action != 'SKIP':
                size_groups[entry.file_size].append(entry)
        
        # 过滤掉唯一大小的文件
        potential_dups = [group for group in size_groups.values() if len(group) > 1]
        
        # 第二级：快速哈希（头+中+尾 64KB，MD5）
        quick_hash_groups = defaultdict(list)
        for group in potential_dups:
            for entry in group:
                qhash = self._quick_hash(entry.source_path)
                quick_hash_groups[qhash].append(entry)
        
        potential_dups = [group for group in quick_hash_groups.values() if len(group) > 1]
        
        # 第三级：完整哈希（SHA-256）
        full_hash_groups = defaultdict(list)
        for group in potential_dups:
            for entry in group:
                fhash = self._full_hash(entry.source_path)
                full_hash_groups[fhash].append(entry)
        
        # 标记重复文件
        for fhash, group in full_hash_groups.items():
            if len(group) > 1:
                keep = self._select_keep(group)
                for entry in group:
                    if entry is not keep:
                        entry.action = 'SKIP'
                        entry.action_reason = f'duplicate_of:{keep.source_path}'
        
        return self.entries
    
    def _quick_hash(self, path: Path) -> str:
        """计算快速哈希：头+中+尾各64KB"""
        size = path.stat().st_size
        chunk_size = 64 * 1024
        
        with open(path, 'rb') as f:
            # 头部
            head = f.read(chunk_size)
            # 中间
            f.seek(max(0, size // 2 - chunk_size // 2))
            middle = f.read(chunk_size)
            # 尾部
            f.seek(max(0, size - chunk_size))
            tail = f.read(chunk_size)
        
        combined = head + middle + tail
        return hashlib.md5(combined).hexdigest()
    
    def _full_hash(self, path: Path) -> str:
        """计算完整SHA-256"""
        sha256 = hashlib.sha256()
        with open(path, 'rb') as f:
            while True:
                chunk = f.read(8 * 1024 * 1024)  # 8MB chunks
                if not chunk:
                    break
                sha256.update(chunk)
        return sha256.hexdigest()
    
    def _select_keep(self, group: List[MapEntry]) -> MapEntry:
        """根据策略选择保留的文件"""
        if self.strategy == 'keep_first':
            return group[0]
        elif self.strategy == 'keep_largest':
            return max(group, key=lambda e: e.file_size)
        elif self.strategy == 'keep_shortest_path':
            return min(group, key=lambda e: len(e.source_path.parts))
        else:
            return group[0]
```

```python
# safety/file_executor.py
class FileExecutor:
    def __init__(self, mode: str = 'auto', workers: int = 8):
        self.mode = mode
        self.workers = workers
        self.error_collector = ErrorCollector()
    
    def execute(self, entries: List[MapEntry]) -> ExecutionStats:
        """执行物理重组"""
        # 判断同卷/跨卷
        if self.mode == 'auto':
            self.mode = self._detect_mode(entries)
        
        stats = ExecutionStats()
        
        with ThreadPoolExecutor(max_workers=self.workers) as executor:
            futures = []
            for entry in entries:
                if entry.action == 'COPY':
                    future = executor.submit(self._copy_or_move, entry)
                    futures.append(future)
            
            for future in as_completed(futures):
                try:
                    result = future.result()
                    stats.update(result)
                except Exception as e:
                    self.error_collector.add(e)
        
        return stats
    
    def _detect_mode(self, entries: List[MapEntry]) -> str:
        """自动检测同卷/跨卷"""
        if not entries:
            return 'copy'
        
        first_source = entries[0].source_path
        first_target = entries[0].target_path
        
        # 检查盘符
        if first_source.drive != first_target.drive:
            return 'copy'
        
        # 同盘符，尝试用 os.rename 测试（快速判断是否同卷）
        try:
            # 创建临时文件测试
            test_file = first_source.parent / '.nesttidy_test'
            test_file.touch()
            test_target = first_target.parent / '.nesttidy_test'
            os.rename(test_file, test_target)
            os.rename(test_target, test_file)
            test_file.unlink()
            return 'move'
        except OSError:
            return 'copy'
    
    def _copy_or_move(self, entry: MapEntry) -> dict:
        """带重试的复制/移动"""
        max_retries = 3
        retry_delay = 0.5
        
        for attempt in range(max_retries):
            try:
                if self.mode == 'copy':
                    return self._copy_with_retry(entry)
                elif self.mode == 'move':
                    return self._move_with_retry(entry)
            except (OSError, PermissionError) as e:
                if attempt < max_retries - 1:
                    time.sleep(retry_delay)
                    continue
                raise
        
        raise RuntimeError("Unreachable")
    
    def _copy_with_retry(self, entry: MapEntry) -> dict:
        """带进度报告的复制"""
        # 创建目标目录
        entry.target_path.parent.mkdir(parents=True, exist_ok=True)
        
        # 大文件显示进度
        if entry.file_size > 50 * 1024 * 1024:  # 50MB
            return self._copy_with_progress(entry)
        else:
            shutil.copy2(entry.source_path, entry.target_path)
            return {'action': 'copy', 'size': entry.file_size}
    
    def _move_with_retry(self, entry: MapEntry) -> dict:
        """移动（同卷rename或跨卷copy+delete）"""
        entry.target_path.parent.mkdir(parents=True, exist_ok=True)
        
        try:
            # 尝试 rename（同卷）
            os.rename(entry.source_path, entry.target_path)
            return {'action': 'move_rename', 'size': 0}
        except OSError:
            # 跨卷，降级为 copy + delete
            shutil.copy2(entry.source_path, entry.target_path)
            
            # 校验（可选）
            if self._verify_copy(entry):
                entry.source_path.unlink()
                return {'action': 'move_copy', 'size': entry.file_size}
            else:
                raise RuntimeError(f"校验失败: {entry.source_path}")
    
    def _verify_copy(self, entry: MapEntry) -> bool:
        """跨卷移动时校验文件一致性"""
        # 使用快速哈希校验
        source_hash = DupDetector._quick_hash(entry.source_path)
        target_hash = DupDetector._quick_hash(entry.target_path)
        return source_hash == target_hash
```

```python
class DiffGenerator:
    def __init__(self, source_tree: FileNode, target_tree: FileNode):

    def generate_html(self, output_path: Path):
        """生成 HTML 可视化对比报告"""
        from jinja2 import Template
        
        # 构建差异树
        diff_tree = self._build_diff_tree(self.source, self.target)
        
        # 渲染模板
        template = Template(HTML_TEMPLATE)
        html = template.render(
            source_name=self.source.name,
            target_name=self.target.name,
            diff_tree=diff_tree,
            timestamp=datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        )
        
        output_path.write_text(html, encoding='utf-8')
    
    def _build_diff_tree(self, source: FileNode, target: FileNode) -> dict:
        """递归比对源树和目标树"""
        diff = {
            'name': source.name,
            'type': 'folder' if not source.is_file else 'file',
            'status': 'unchanged',
            'children': [],
            'details': None
        }
        
        if source.is_file:
            # 文件节点
            if not self._file_exists_in_target(source, target):
                diff['status'] = 'removed'
        else:
            # 文件夹节点
            source_children = {c.name.lower(): c for c in source.children}
            target_children = {c.name.lower(): c for c in target.children if c.name.lower() in source_children}
            
            for name_lower, source_child in source_children.items():
                if name_lower in target_children:
                    target_child = target_children[name_lower]
                    child_diff = self._build_diff_tree(source_child, target_child)
                    diff['children'].append(child_diff)
                else:
                    # 在目标中消失
                    if source_child.is_file:
                        diff['children'].append({
                            'name': source_child.name,
                            'type': 'file',
                            'status': 'removed',
                            'children': [],
                            'details': '被穿透消失'
                        })
                    else:
                        diff['children'].append({
                            'name': source_child.name,
                            'type': 'folder',
                            'status': 'removed',
                            'children': [],
                            'details': '纯空壳节点消失'
                        })
            
            # 检查重命名
            for child in target.children:
                if child.safe_name != child.name:
                    diff['children'].append({
                        'name': child.safe_name,
                        'type': 'file' if child.is_file else 'folder',
                        'status': 'renamed',
                        'children': [],
                        'details': f'原名: {child.name}'
                    })
        
        return diff
```

```python
        self.throttle_ms = throttle_ms
        self.counter = 0
```

```python
class ErrorCollector:
```

```python
# safety/pre_checker.py
class SafetyChecker:
    # 系统目录保护列表
    PROTECTED_PATHS = [
        Path('C:/Windows'),
        Path('C:/Program Files'),
        Path('C:/Program Files (x86)')
    ]
    
    def check_safety(self, source: Path, target: Path) -> SafetyLevel:
        """五层安全检查"""
        # 第一层：系统目录保护
        for protected in self.PROTECTED_PATHS:
            if self._is_subpath(source, protected) or self._is_subpath(target, protected):
                return SafetyLevel.BLOCKED
        
        # 第二层：路径安全检查
        if not self._check_path_safety(source, target):
            return SafetyLevel.RESTRICTED
        
        # 第三层：写入权限检查
        if not self._check_write_permission(target):
            return SafetyLevel.RESTRICTED
        
        # 第四层：磁盘空间检查
        if not self._check_disk_space(source, target):
            return SafetyLevel.RESTRICTED
        
        return SafetyLevel.FULL
    
    def _is_subpath(self, path: Path, parent: Path) -> bool:
        """检查 path 是否是 parent 的子路径"""
        try:
            path.relative_to(parent)
            return True
        except ValueError:
            return False
```

```python
# safety/lock_manager.py
class LockManager:
    def __init__(self, target_dir: Path):
        self.lock_file = target_dir / '.restructure.lock'
        self.session_id = str(uuid.uuid4())
    
    def acquire(self, force: bool = False) -> bool:
        """获取锁"""
        if self.lock_file.exists():
            if not force:
                existing = json.loads(self.lock_file.read_text())
                raise RuntimeError(f"目录被锁定: PID={existing['pid']}, 时间={existing['start_time']}")
            else:
                self.lock_file.unlink()
        
        lock_data = {
            'pid': os.getpid(),
            'hostname': socket.gethostname(),
            'username': os.getlogin(),
            'session_id': self.session_id,
            'start_time': datetime.now().isoformat(),
            'expire_time': (datetime.now() + timedelta(hours=12)).isoformat(),
            'command_line': ' '.join(sys.argv)
        }
        
        self.lock_file.write_text(json.dumps(lock_data, indent=2))
        
        # 注册 atexit 清理
        atexit.register(self.release)
        
        return True
    
    def release(self):
        """释放锁"""
        if self.lock_file.exists():
            try:
                existing = json.loads(self.lock_file.read_text())
                if existing['session_id'] == self.session_id:
                    self.lock_file.unlink()
            except:
                pass
```

```python
        script.append(']')
        script.append('')
        script.append('def rollback():')
        script.append('    for op in operations:')
        script.append('        try:')
        script.append('            if op["action"] == "copy":')

        script.append('                if target.exists():')

        script.append('        except Exception as e:')
        script.append('            print(f"回滚失败: {op} - {e}")')
        script.append('')
        script.append('if __name__ == "__main__":')
        script.append('    print("开始回滚...")')
        script.append('    rollback()')
        script.append('    print("回滚完成")')
        
        output_path.write_text('\n'.join(script), encoding='utf-8')
```
